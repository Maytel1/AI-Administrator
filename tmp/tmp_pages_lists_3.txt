 828 | def _trend(diff: dict[str, Any] | None, good_down: bool = False) -> str:
 829 |     """Изменение к прошлому периоду под плиткой: «+100.0 % к прошлому».
 830 | 
 831 |     ``good_down=True`` — для показателей, где спад это хорошо (отмены, долг).
 832 |     Без ненулевой базы (было 0) процента нет — пишем «не с чем сравнить»,
 833 |     а не «+∞»: деление на ноль это не рост.
 834 |     """
 835 |     if not isinstance(diff, dict):
 836 |         return ""
 837 |     percent = diff.get("percent")
 838 |     if percent is None:
 839 |         return '<i class="trend flat">не с чем сравнить</i>'
 840 |     if percent == 0:
 841 |         return '<i class="trend flat">0.0 % к прошлому</i>'
 842 |     up = percent > 0
 843 |     good = up != good_down
 844 |     return (
 845 |         f'<i class="trend {"up" if good else "down"}">'
 846 |         f"{percent:+.1f} % к прошлому</i>"
 847 |     )

 870 | def _compare_table(
